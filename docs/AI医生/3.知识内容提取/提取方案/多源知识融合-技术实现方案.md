#### 3.1.1 优先级层次获取

```python
def get_priority_level(source: SourceKnowledge) -> int:
    """获取来源的优先级层次"""
    priority_order = {
        "national_guideline": 1,
        "international_guideline": 2,
        "authoritative_textbook": 3,
        "professional_textbook": 4,
        "expert_consensus": 5,
        "academic_paper": 6,
        "other": 7
    }
    return priority_order.get(source.source_type, 7)
```

```python
def compare_authority(
    source1: SourceKnowledge, 
    source2: SourceKnowledge
) -> str:
    """
    比较两个来源的权威性
    返回: "source1_higher", "source2_higher", "equal", "need_expert_review"
    """
    
    # 规则1：优先级层次比较
    priority1 = get_priority_level(source1)
    priority2 = get_priority_level(source2)
    
    # 如果优先级层次不同，直接返回
    if priority1 < priority2:
        return "source1_higher"
    elif priority1 > priority2:
        return "source2_higher"
    
    # 如果优先级层次相同，进入细化规则判断
    return compare_same_priority_sources(source1, source2)

def compare_same_priority_sources(
    source1: SourceKnowledge, 
    source2: SourceKnowledge
) -> str:
    """比较同优先级来源的权威性"""
    
    # 规则2：时间优先（对于指南类）
    if source1.source_type in ["national_guideline", "international_guideline"]:
        if source1.publication_year > source2.publication_year:
            # 如果时间差异超过3年，新版本优先
            if source1.publication_year - source2.publication_year >= 3:
                return "source1_higher"
        elif source2.publication_year > source1.publication_year:
            if source2.publication_year - source1.publication_year >= 3:
                return "source2_higher"
        # 如果时间差异小于3年，需要专家判断
        return "need_expert_review"
    
    # 规则3：版本优先（对于教材类）
    if source1.source_type in ["authoritative_textbook", "professional_textbook"]:
        if source1.is_latest_version and not source2.is_latest_version:
            return "source1_higher"
        elif source2.is_latest_version and not source1.is_latest_version:
            return "source2_higher"
        # 如果都是最新版本或都不是，需要进一步判断
        return "need_expert_review"
    
    # 规则4：其他情况需要专家判断
    return "need_expert_review"
```

#### 3.1.3 来源排序

```python
def sort_sources_by_authority(
    sources: List[SourceKnowledge]
) -> List[SourceKnowledge]:
    """基于规则对来源进行权威性排序"""
    
    def get_sort_key(source: SourceKnowledge) -> tuple:
        """生成排序键"""
        priority = get_priority_level(source)
        
        # 第一层：优先级
        # 第二层：时间（越新越好，用负数）
        # 第三层：是否最新版本（True在前）
        return (
            priority,
            -source.publication_year,  # 负数，使新版本在前
            not source.is_latest_version  # False在前（最新版本优先）
        )
    
    return sorted(sources, key=get_sort_key)
```

def detect_value_conflicts(
    sources: List[SourceKnowledge]
) -> List[Conflict]:
    """检测值冲突"""
    conflicts = []
    
    # 遍历所有字段
    for field_name in UNIFIED_TEMPLATE_FIELDS:
        values = {}
        for source in sources:
            if field_name in source.content:
                value = source.content[field_name]
                if value not in values:
                    values[value] = []
                values[value].append(source)
        
        # 如果同一字段有多个不同的值，则存在冲突
        if len(values) > 1:
            conflict = Conflict(
                conflict_id=f"value_conflict_{field_name}_{datetime.now().timestamp()}",
                conflict_type="value_conflict",
                field_name=field_name,
                conflicting_sources=[s.source_id for sources_list in values.values() for s in sources_list],
                conflicting_values=list(values.keys()),
                severity=calculate_severity(field_name),
                detected_at=datetime.now()
            )
            conflicts.append(conflict)
    
    return conflicts

def calculate_severity(field_name: str) -> str:
    """计算字段冲突的严重程度"""
    high_severity_fields = ["diagnosis_criteria", "treatment_principle", "drug_dosage"]
    medium_severity_fields = ["clinical_manifestation", "etiology"]
    
    if field_name in high_severity_fields:
        return "high"
    elif field_name in medium_severity_fields:
        return "medium"
    else:
        return "low"
```

from enum import Enum

class FusionStrategy(Enum):
    AUTHORITY_PRIORITY = "authority_priority"
    LATEST_PRIORITY = "latest_priority"
    COMPLEMENTARY = "complementary"
    PROCESS_SELECTION = "process_selection"
    SEGMENTED = "segmented"
    MANUAL = "manual"

def select_fusion_strategy(
    sources: List[SourceKnowledge],
    conflicts: List[Conflict]
) -> FusionStrategy:
    """选择融合策略"""
    
    # 1. 如果没有冲突，使用互补融合
    if not conflicts:
        return FusionStrategy.COMPLEMENTARY
    
    # 2. 检查流程完整性冲突
    process_conflicts = [c for c in conflicts if c.conflict_type == "process_integrity_conflict"]
    if process_conflicts:
        # 检查是否可以分段融合
        if can_segmented_fusion(sources, process_conflicts):
            return FusionStrategy.SEGMENTED
        else:
            # 不能分段融合，选择完整流程
            return FusionStrategy.PROCESS_SELECTION
    
    # 3. 检查值冲突
    value_conflicts = [c for c in conflicts if c.conflict_type == "value_conflict"]
    if value_conflicts:
        # 检查优先级层次是否不同
        source_priorities = [get_priority_level(s) for s in sources]
        min_priority = min(source_priorities)
        max_priority = max(source_priorities)
        
        # 如果优先级层次不同，使用权威优先策略
        if min_priority != max_priority:
            return FusionStrategy.AUTHORITY_PRIORITY
        else:
            # 优先级层次相同，检查时间差异
            years = [s.publication_year for s in sources]
            time_diff = max(years) - min(years)
            if time_diff >= 3:  # 3年以上
                return FusionStrategy.LATEST_PRIORITY
            else:
                # 需要人工判断
                return FusionStrategy.MANUAL
    
    # 4. 检查形式差异冲突
    format_conflicts = [c for c in conflicts if c.conflict_type == "format_difference_conflict"]

            return FusionStrategy.COMPLEMENTARY
        else:
            return FusionStrategy.MANUAL
    
    # 5. 默认使用互补融合
    return FusionStrategy.COMPLEMENTARY

def authority_priority_fusion(
    sources: List[SourceKnowledge],
    conflicts: List[Conflict]
) -> FusedKnowledge:
    """权威优先融合（基于优先级规则）"""
    
    # 按优先级规则排序
    sorted_sources = sort_sources_by_authority(sources)
    primary_source = sorted_sources[0]
    
    # 以最权威的来源为主
    fused_content = primary_source.content.copy()
    
    # 其他来源作为补充
    supplementary_sources = sorted_sources[1:]
    supplementary_info = []
    for source in supplementary_sources:
        supplementary_info.append({
            "source_id": source.source_id,
            "source_name": source.source_name,
            "differences": find_differences(primary_source.content, source.content)
        })
    
    return FusedKnowledge(
        knowledge_id=primary_source.knowledge_id,
        fused_content=fused_content,
        fusion_strategy="authority_priority",
        primary_source=primary_source.source_id,
        supplementary_sources=[s.source_id for s in supplementary_sources],
        fusion_metadata={
            "supplementary_info": supplementary_info
        }
    )

def find_differences(content1: Dict, content2: Dict) -> Dict:
    """找出两个内容之间的差异"""
    differences = {}
    all_keys = set(content1.keys()) | set(content2.keys())
    
    for key in all_keys:
        val1 = content1.get(key)
        val2 = content2.get(key)
        if val1 != val2:
            differences[key] = {
                "value1": val1,
                "value2": val2
            }
    
    return differences
```

def process_selection_fusion(
    sources: List[SourceKnowledge],
    conflicts: List[Conflict]
) -> FusedKnowledge:
    """流程选择融合（不能拼接，选择完整流程）"""
    
    # 按优先级规则排序来源
    sorted_sources = sort_sources_by_authority(sources)
    
    # 选择优先级最高且流程完整的来源
    best_source = None
    best_process = None
    for source in sorted_sources:
        process = source.content.get("diagnosis_process")
        if process and is_process_complete(process):
            best_source = source
            best_process = process
            break
    
    # 如果没有找到完整流程，选择第一个
    if not best_source:
        best_source = sorted_sources[0]
        best_process = best_source.content.get("diagnosis_process")
    
    # 以最优流程为主
    fused_content = best_source.content.copy()
    fused_content["diagnosis_process"] = best_process
    
    # 其他流程作为备选方案
    alternative_processes = []
    for source in sorted_sources:
        if source != best_source:
            process = source.content.get("diagnosis_process")
            if process:
                alternative_processes.append({
                    "source_id": source.source_id,
                    "source_name": source.source_name,
                    "process": process,
                    "priority": get_priority_level(source)
                })
    
    return FusedKnowledge(
        knowledge_id=best_source.knowledge_id,
        fused_content=fused_content,
        fusion_strategy="process_selection",
        primary_source=best_source.source_id,

def verify_process_integrity(process: Dict) -> bool:
    """验证流程完整性"""
    
    # 1. 检查是否有开始和结束
    if "start" not in process or "end" not in process:
        return False
    
    # 2. 检查步骤之间的逻辑连贯性
    steps = process.get("steps", [])
    for i in range(len(steps) - 1):
        current_step = steps[i]
        next_step = steps[i + 1]
        
        # 检查当前步骤的输出是否与下一步骤的输入匹配
        if not check_step_connection(current_step, next_step):
            return False
    
    # 3. 检查步骤依赖关系
    dependencies = process.get("dependencies", {})
    for step_id, deps in dependencies.items():
        for dep_id in deps:
            if dep_id not in [s["id"] for s in steps]:
                return False
    
    return True

def calculate_completeness_score(
