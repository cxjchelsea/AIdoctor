# 知识图谱搭建实施指南

> **目标**：基于总体结构设计，提供正确的知识图谱搭建实施步骤  
> **核心原则**：**先结构，再节点，后关系**  
> **创建日期**：2026-01-28  
> **文档版本**：v2.0

---

## 📋 实施总览

### 正确的实施顺序

**三步走策略**：

1. **第一步：搭建结构（Schema）** ⭐⭐⭐⭐⭐
   - 定义所有节点类型（Node Labels）
   - 定义所有关系类型（Relationship Types）
   - 定义属性结构（Properties）
   - 创建约束（Constraints）和索引（Indexes）

2. **第二步：创建节点实例** ⭐⭐⭐⭐⭐
   - 创建具体的节点（如：Disease节点、Symptom节点、Category节点等）
   - 设置节点属性

3. **第三步：建立关系** ⭐⭐⭐⭐⭐
   - 建立节点之间的关系（如：BELONGS_TO、SUPPORTED_BY等）
   - 设置关系属性

### 关于分类的正确理解

**分类（Category）的正确位置**：
- **阶段1**：定义Category节点类型和BELONGS_TO关系类型（结构定义）
- **阶段2**：创建Category节点实例（创建节点）
- **阶段3**：建立HAS_CATEGORY和BELONGS_TO关系（建立关系）

**分类不是第一步，而是结构的一部分**：
- Category是节点类型的一种，在阶段1定义
- BELONGS_TO是关系类型的一种，在阶段1定义
- Category节点在阶段2创建
- BELONGS_TO关系在阶段3建立

### 阶段1：搭建知识图谱结构（Schema定义）⭐⭐⭐⭐⭐

**目标**：定义所有节点类型、关系类型、属性结构、约束和索引

**时间估算**：1-2天

**为什么先做**：
- 结构是基础，必须先定义好才能创建节点和关系
- 约束可以保证数据完整性（如display_id唯一性）
- 索引可以提升查询性能
- 结构定义是一次性的，后续只需要创建节点和关系

**关系类型列表**（基于总体结构设计）：

```cypher
// - DIFFERENTIAL_WITH（疾病需要与另一疾病鉴别）
```

```python
# scripts/create_categories.py
from neo4j import GraphDatabase
from datetime import datetime

def create_categories(uri, user, password, csv_file):
    driver = GraphDatabase.driver(uri, auth=(user, password))
    
    with driver.session() as session:
        categories = read_csv(csv_file)
        
        # 创建Category节点（只创建节点，不建立关系）
        for cat in categories:
            session.run("""
                CREATE (c:Category {
                    category_uuid: $uuid,
                    display_id: $display_id,
                    name: $name,
                    category_type: $category_type,
                    parent_category: $parent_category,
                    level: $level,
                    description: $description,
                    created_at: datetime(),
                    updated_at: datetime(),
                    version: 'v1.0'
                })
            """, 
                uuid=cat['category_uuid'],
                display_id=cat['display_id'],
                name=cat['name'],
                category_type=cat['category_type'],
                parent_category=cat.get('parent_category', ''),
                level=int(cat['level']),
                description=cat.get('description', '')
            )
    
    driver.close()
```

1. **建立HAS_CATEGORY关系（分类层级）**

```cypher
// 建立分类层级关系
MATCH (parent:Category), (child:Category)
WHERE child.parent_category = parent.name
CREATE (parent)-[:HAS_CATEGORY {
    level: child.level - parent.level,
    confidence: 1.0,
    source_doc_id: 'manual',
    source_chunk_id: ['chunk_manual'],
    extraction_method: 'manual',
    created_at: datetime(),
    version: 'v1.0'
```

2. **建立BELONGS_TO关系（实体属于分类）**

```python
# scripts/create_belongs_to_relations.py
def create_belongs_to_relations(uri, user, password):
    driver = GraphDatabase.driver(uri, auth=(user, password))
    
    with driver.session() as session:
        # 为Symptom建立BELONGS_TO关系
        session.run("""
            MATCH (s:Symptom), (c:Category)
            WHERE s.category = c.name
            CREATE (s)-[:BELONGS_TO {
                confidence: 1.0,
                source_doc_id: 'manual',
                source_chunk_id: ['chunk_manual'],
                extraction_method: 'manual',
                created_at: datetime(),
                version: 'v1.0'
            }]->(c)
        """)
        
        # 为Disease建立BELONGS_TO关系
        session.run("""
            MATCH (d:Disease), (c:Category)
            WHERE d.category = c.name
            CREATE (d)-[:BELONGS_TO {
                confidence: 1.0,
                source_doc_id: 'manual',
                source_chunk_id: ['chunk_manual'],
                extraction_method: 'manual',
                created_at: datetime(),
                version: 'v1.0'
            }]->(c)
        """)
        
        # 为Examination建立BELONGS_TO关系
        session.run("""
            MATCH (e:Examination), (c:Category)
            WHERE e.category = c.name

                confidence: 1.0,
                source_doc_id: 'manual',
```

```python
# scripts/clean_data.py
import pandas as pd
from datetime import datetime
import uuid

def clean_categories(csv_file):
    """清洗分类数据"""
    df = pd.read_csv(csv_file)
    
    # 1. 去除前后空格
    df = df.apply(lambda x: x.str.strip() if x.dtype == "object" else x)
    
    # 2. 补充UUID（如果缺失）
    if 'category_uuid' not in df.columns or df['category_uuid'].isna().any():
        df['category_uuid'] = df.apply(lambda x: str(uuid.uuid4()) if pd.isna(x.get('category_uuid', '')) else x['category_uuid'], axis=1)
    
    # 3. 补充时间戳
    if 'created_at' not in df.columns:
        df['created_at'] = datetime.now().isoformat()
    if 'updated_at' not in df.columns:
        df['updated_at'] = datetime.now().isoformat()
    
    # 4. 补充版本号
    if 'version' not in df.columns:
        df['version'] = 'v1.0'
    
    # 5. 验证必填字段
    required_fields = ['category_uuid', 'display_id', 'name', 'category_type', 'level']
    for field in required_fields:
        if field not in df.columns or df[field].isna().any():
            raise ValueError(f"必填字段 {field} 缺失或为空")
    
    # 6. 验证唯一性
    if df['display_id'].duplicated().any():
        raise ValueError("display_id 存在重复")
    
    # 7. 保存清洗后的数据
    df.to_csv(csv_file.replace('.csv', '_cleaned.csv'), index=False, encoding='utf-8-sig')
    
    return df
```

```python
def create_categories(uri, user, password, csv_file):

            categories = read_csv(csv_file)

            def create_category_batch(tx, batch):
                for cat in batch:
                    tx.run("""

                            category_uuid: $uuid,
                            display_id: $display_id,
                            name: $name,
                            category_type: $category_type,
                            parent_category: $parent_category,
                            level: $level,
                            description: $description,

                            updated_at: datetime(),
                            version: 'v1.0'
                        })
                    """, **cat)
            
            # 批量处理（每批100条）
            batch_size = 100
            for i in range(0, len(categories), batch_size):
                batch = categories[i:i+batch_size]
                session.write_transaction(create_category_batch, batch)
                print(f"已导入 {min(i+batch_size, len(categories))}/{len(categories)} 条数据")
    
    except ServiceUnavailable as e:
        print(f"Neo4j服务不可用: {e}")

        raise
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
知识图谱结构搭建脚本
功能：创建所有约束和索引
"""

from neo4j import GraphDatabase
import sys

def setup_schema(uri, user, password):
    """创建所有约束和索引"""
    driver = GraphDatabase.driver(uri, auth=(user, password))
    
    try:
        with driver.session() as session:
            print("开始创建约束和索引...")
            
            # 1. 创建display_id唯一性约束
            constraints = [
                ("symptom_display_id", "Symptom", "display_id"),
                ("chief_complaint_display_id", "ChiefComplaint", "display_id"),
                ("disease_display_id", "Disease", "display_id"),
                ("examination_display_id", "Examination", "display_id"),
                ("category_display_id", "Category", "display_id"),
                ("evidence_display_id", "Evidence", "display_id"),
                ("reasoning_subgroup_display_id", "ReasoningSubgroup", "display_id"),
                ("differential_framework_subclass_display_id", "DifferentialFrameworkSubclass", "display_id"),
                ("differential_point_display_id", "DifferentialPoint", "display_id"),
            ]
            
            for constraint_name, label, field in constraints:
                try:
                    session.run(f"""
                        CREATE CONSTRAINT {constraint_name} IF NOT EXISTS
                        FOR (n:{label}) REQUIRE n.{field} IS UNIQUE
                    """)
                    print(f"✅ 创建约束: {constraint_name}")
                except Exception as e:
                    print(f"⚠️  约束 {constraint_name} 创建失败: {e}")
            
            # 2. 创建UUID唯一性约束
            uuid_constraints = [
                ("symptom_uuid", "Symptom", "symptom_uuid"),
                ("chief_complaint_uuid", "ChiefComplaint", "chief_complaint_uuid"),
                ("disease_uuid", "Disease", "disease_uuid"),
                ("examination_uuid", "Examination", "examination_uuid"),
                ("category_uuid", "Category", "category_uuid"),
                ("reasoning_subgroup_uuid", "ReasoningSubgroup", "subgroup_uuid"),
                ("differential_framework_subclass_uuid", "DifferentialFrameworkSubclass", "subclass_uuid"),
                ("differential_point_uuid", "DifferentialPoint", "differential_point_uuid"),
            ]
            
            for constraint_name, label, field in uuid_constraints:
                try:
                    session.run(f"""
                        CREATE CONSTRAINT {constraint_name} IF NOT EXISTS
                        FOR (n:{label}) REQUIRE n.{field} IS UNIQUE
                    """)
                    print(f"✅ 创建UUID约束: {constraint_name}")
                except Exception as e:
                    print(f"⚠️  UUID约束 {constraint_name} 创建失败: {e}")
            
            # 3. 创建索引
            indexes = [
                ("symptom_name", "Symptom", "name"),
                ("disease_name", "Disease", "name"),
                ("disease_category", "Disease", "category"),
                ("disease_icd10", "Disease", "icd10_code"),
                ("disease_icd11", "Disease", "icd11_code"),
                ("disease_tier_level", "Disease", "tier_level"),
                ("chief_complaint_name", "ChiefComplaint", "name"),
            ]
            
            for index_name, label, field in indexes:
                try:
                    session.run(f"""
                        CREATE INDEX {index_name} IF NOT EXISTS
                        FOR (n:{label}) ON (n.{field})
                    """)
                    print(f"✅ 创建索引: {index_name}")
                except Exception as e:
                    print(f"⚠️  索引 {index_name} 创建失败: {e}")
            
            # 4. 验证
            result = session.run("SHOW CONSTRAINTS")
            constraint_count = len(list(result))
            print(f"\n✅ 共创建 {constraint_count} 个约束")
            
            result = session.run("SHOW INDEXES")
            index_count = len(list(result))
            print(f"✅ 共创建 {index_count} 个索引")
            
            print("\n✅ 结构搭建完成！")
    
    except Exception as e:
        print(f"❌ 结构搭建失败: {e}")
        sys.exit(1)
    finally:
        driver.close()

if __name__ == "__main__":
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
节点创建脚本（通用版）
支持创建：Category, Symptom, Disease, ChiefComplaint, Examination
"""

from neo4j import GraphDatabase
import pandas as pd
import uuid
from datetime import datetime
import sys

def read_csv(file_path):
    """读取CSV文件"""
    try:
        df = pd.read_csv(file_path, encoding='utf-8-sig')
        return df.to_dict('records')
    except Exception as e:
        print(f"❌ 读取CSV文件失败: {e}")
        sys.exit(1)

def create_nodes(uri, user, password, csv_file, node_type):
    """批量创建节点"""
    driver = GraphDatabase.driver(uri, auth=(user, password))
    
    try:
        nodes = read_csv(csv_file)
        print(f"读取到 {len(nodes)} 条{node_type}数据")
        
        # 数据清洗和补全
        for node in nodes:
            # 补全UUID（如果缺失）
            uuid_field = f"{node_type.lower()}_uuid"
            if uuid_field not in node or pd.isna(node.get(uuid_field)):
                node[uuid_field] = str(uuid.uuid4())
            
            # 补全时间戳（如果缺失）
            if 'created_at' not in node or pd.isna(node.get('created_at')):
                node['created_at'] = datetime.now().isoformat()
            if 'updated_at' not in node or pd.isna(node.get('updated_at')):
                node['updated_at'] = datetime.now().isoformat()
            
            # 补全版本号（如果缺失）
            if 'version' not in node or pd.isna(node.get('version')):
                node['version'] = 'v1.0'
        
        with driver.session() as session:
            # 使用UNWIND批量创建（性能优化）
            def create_batch(tx, batch):
                # 构建属性映射
                props = {}
                for node in batch:
                    for key, value in node.items():
                        if key not in props:
                            props[key] = []
                        props[key].append(value if not pd.isna(value) else None)
                
                # 构建CREATE语句
                create_props = []
                for key in props.keys():
                    if key.endswith('_uuid'):
                        prop_name = key
                    else:
                        prop_name = key
                    create_props.append(f"{prop_name}: node.{prop_name}")
                
                query = f"""
                UNWIND $batch AS node
                CREATE (n:{node_type} {{
                    {', '.join([f'{k}: node.{k}' for k in props.keys()])}
                }})
                RETURN count(n) as count
                """
                
                result = tx.run(query, batch=batch)
                return result.single()['count']
            
            # 批量处理（每批1000条）
            batch_size = 1000
            total_created = 0
            
            for i in range(0, len(nodes), batch_size):
                batch = nodes[i:i+batch_size]
                try:
                    count = session.write_transaction(create_batch, batch)
                    total_created += count
                    print(f"✅ 已创建 {total_created}/{len(nodes)} 个{node_type}节点")
                except Exception as e:
                    print(f"❌ 批量创建失败（第{i//batch_size + 1}批）: {e}")
                    # 记录失败的数据
                    print(f"失败数据: {batch[:3]}...")  # 只打印前3条
                    raise
            
            print(f"\n✅ {node_type}节点创建完成！共创建 {total_created} 个节点")
    
    except Exception as e:
        print(f"❌ 节点创建失败: {e}")
        sys.exit(1)
    finally:
        driver.close()

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description='创建知识图谱节点')
    parser.add_argument('--uri', default='bolt://localhost:7687', help='Neo4j连接URI')
    parser.add_argument('--user', default='neo4j', help='Neo4j用户名')
    parser.add_argument('--password', required=True, help='Neo4j密码')
    parser.add_argument('--csv', required=True, help='CSV文件路径')
    parser.add_argument('--type', required=True, 
                       choices=['Category', 'Symptom', 'Disease', 'ChiefComplaint', 'Examination'],
                       help='节点类型')
    
    args = parser.parse_args()
    create_nodes(args.uri, args.user, args.password, args.csv, args.type)
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
关系建立脚本
支持建立：BELONGS_TO, HAS_CATEGORY等关系
"""

from neo4j import GraphDatabase
from datetime import datetime
import sys

def create_belongs_to_relations(uri, user, password):
    """建立BELONGS_TO关系（实体属于分类）"""
    driver = GraphDatabase.driver(uri, auth=(user, password))
    
    try:
        with driver.session() as session:
            # 为Symptom建立BELONGS_TO关系
            result = session.run("""
                MATCH (s:Symptom), (c:Category)
                WHERE s.category = c.name
                CREATE (s)-[r:BELONGS_TO {
                    confidence: 1.0,
                    source_doc_id: 'manual',
                    source_chunk_id: ['chunk_manual'],
                    extraction_method: 'manual',
                    created_at: datetime(),
                    version: 'v1.0'
                }]->(c)
                RETURN count(r) as count
            """)
            symptom_count = result.single()['count']
            print(f"✅ 为Symptom建立 {symptom_count} 个BELONGS_TO关系")
            
            # 为Disease建立BELONGS_TO关系
            result = session.run("""
                MATCH (d:Disease), (c:Category)
                WHERE d.category = c.name
                CREATE (d)-[r:BELONGS_TO {
                    confidence: 1.0,
                    source_doc_id: 'manual',
                    source_chunk_id: ['chunk_manual'],
                    extraction_method: 'manual',
                    created_at: datetime(),
                    version: 'v1.0'
                }]->(c)
                RETURN count(r) as count
            """)
            disease_count = result.single()['count']
            print(f"✅ 为Disease建立 {disease_count} 个BELONGS_TO关系")
            
            # 为Examination建立BELONGS_TO关系
            result = session.run("""
                MATCH (e:Examination), (c:Category)
                WHERE e.category = c.name
                CREATE (e)-[r:BELONGS_TO {
                    confidence: 1.0,

                    source_chunk_id: ['chunk_manual'],
                    extraction_method: 'manual',
                    created_at: datetime(),
                    version: 'v1.0'
                }]->(c)
                RETURN count(r) as count
            """)
            exam_count = result.single()['count']
            print(f"✅ 为Examination建立 {exam_count} 个BELONGS_TO关系")
            
            print(f"\n✅ BELONGS_TO关系建立完成！共建立 {symptom_count + disease_count + exam_count} 个关系")
    
    except Exception as e:
        print(f"❌ 关系建立失败: {e}")
        sys.exit(1)
    finally:
        driver.close()

def create_has_category_relations(uri, user, password):
    """建立HAS_CATEGORY关系（分类层级）"""
    driver = GraphDatabase.driver(uri, auth=(user, password))
    
    try:
        with driver.session() as session:
            result = session.run("""
                MATCH (parent:Category), (child:Category)
                WHERE child.parent_category = parent.name
                CREATE (parent)-[r:HAS_CATEGORY {
                    level: child.level - parent.level,
                    confidence: 1.0,
                    source_doc_id: 'manual',
                    source_chunk_id: ['chunk_manual'],
                    extraction_method: 'manual',
                    created_at: datetime(),
                    version: 'v1.0'
                }]->(child)
                RETURN count(r) as count
            """)
            count = result.single()['count']
            print(f"✅ 建立 {count} 个HAS_CATEGORY关系")
    
    except Exception as e:
        print(f"❌ 关系建立失败: {e}")
        sys.exit(1)
    finally:
        driver.close()

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description='建立知识图谱关系')
    parser.add_argument('--uri', default='bolt://localhost:7687', help='Neo4j连接URI')
    parser.add_argument('--user', default='neo4j', help='Neo4j用户名')
    parser.add_argument('--password', required=True, help='Neo4j密码')
    parser.add_argument('--relation', required=True,
                       choices=['BELONGS_TO', 'HAS_CATEGORY'],
                       help='关系类型')
    
    args = parser.parse_args()
    
    if args.relation == 'BELONGS_TO':
        create_belongs_to_relations(args.uri, args.user, args.password)
    elif args.relation == 'HAS_CATEGORY':
        create_has_category_relations(args.uri, args.user, args.password)
```
