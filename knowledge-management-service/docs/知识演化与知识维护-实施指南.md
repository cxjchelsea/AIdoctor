```python
from pydantic import BaseModel, Field
from datetime import datetime
from typing import Optional, Dict, Any
from enum import Enum

class KnowledgeType(str, Enum):
    """知识类型枚举"""
    KG = "kg"           # 知识图谱
    TABLE = "table"     # 表格文件
    CONFIG = "config"   # 配置文件

class VersionStatus(str, Enum):
    """版本状态枚举"""
    DRAFT = "draft"         # 草稿
    ACTIVE = "active"       # 激活
    DEPRECATED = "deprecated"  # 已废弃

class EvidenceLevel(str, Enum):
    """证据等级枚举"""
    A = "A"  # 指南
    B = "B"  # 高质量论文
    C = "C"  # 一般论文

class KnowledgeVersion(BaseModel):
    """知识版本模型"""
    version_id: str = Field(..., description="版本ID")
    knowledge_type: KnowledgeType = Field(..., description="知识类型")
    knowledge_item: int = Field(..., ge=0, le=18, description="知识项编号")
    version_number: int = Field(..., ge=1, description="版本号")
    previous_version_id: Optional[str] = Field(None, description="上一版本ID")
    created_at: datetime = Field(default_factory=datetime.now, description="创建时间")
    created_by: str = Field(..., description="创建者")
    change_summary: str = Field(..., description="变更摘要")
    change_details: Dict[str, Any] = Field(default_factory=dict, description="变更详情")
    source: str = Field(..., description="数据来源")
    evidence_level: EvidenceLevel = Field(..., description="证据等级")
    status: VersionStatus = Field(default=VersionStatus.DRAFT, description="状态")
    metadata: Dict[str, Any] = Field(default_factory=dict, description="元数据")

    class Config:
        json_schema_extra = {
            "example": {
                "version_id": "KNOWLEDGE_v20260128_001_5",
                "knowledge_type": "kg",
                "knowledge_item": 5,
                "version_number": 1,
                "previous_version_id": None,

                "created_by": "admin",
                "change_summary": "新增主诉模板",
                "change_details": {},
                "source": "指南v2026",
                "evidence_level": "A",
                "status": "active",
                "metadata": {}
            }
        }
```

### 2.2 版本管理实现

#### 2.2.1 版本管理器

```python
# app/evolution/version_manager.py
from typing import Optional, List
from datetime import datetime
from app.models.evolution import KnowledgeVersion, VersionStatus
from app.utils.neo4j_client import Neo4jClient
from app.utils.logger import get_logger

logger = get_logger(__name__)

class VersionManager:
    """版本管理器"""
    
    def __init__(self, neo4j_client: Neo4jClient):
        self.neo4j = neo4j_client
    
    def generate_version_id(self, knowledge_type: str, knowledge_item: int, 
                           version_number: int) -> str:
        """生成版本ID"""
        timestamp = datetime.now().strftime("%Y%m%d")
        return f"KNOWLEDGE_v{timestamp}_{version_number:03d}_{knowledge_item}"
    
    async def create_version(self, knowledge_type: str, knowledge_item: int,
                            source: str, change_summary: str, created_by: str,
                            evidence_level: str = "C",
                            previous_version_id: Optional[str] = None) -> KnowledgeVersion:
        """创建新版本"""
        try:
            # 获取当前最大版本号
            max_version = await self.get_max_version_number(knowledge_type, knowledge_item)
            version_number = max_version + 1
            
            # 生成版本ID
            version_id = self.generate_version_id(knowledge_type, knowledge_item, version_number)
            
            # 创建版本对象
            version = KnowledgeVersion(
                version_id=version_id,
                knowledge_type=knowledge_type,
                knowledge_item=knowledge_item,
                version_number=version_number,
                previous_version_id=previous_version_id,
                created_by=created_by,
                change_summary=change_summary,
                source=source,
                evidence_level=evidence_level,
                status=VersionStatus.DRAFT
            )
            
            # 保存到Neo4j
            await self._save_version_to_neo4j(version)
            
            logger.info(f"创建新版本: {version_id}")
            return version
            
        except Exception as e:
            logger.error(f"创建版本失败: {str(e)}")
            raise
    
    async def get_max_version_number(self, knowledge_type: str, 
                                     knowledge_item: int) -> int:
        """获取最大版本号"""
        query = """
        MATCH (v:KnowledgeVersion)
        WHERE v.knowledge_type = $knowledge_type 
          AND v.knowledge_item = $knowledge_item
        RETURN MAX(v.version_number) as max_version
        """
        result = await self.neo4j.execute_query(
            query,
            knowledge_type=knowledge_type,
            knowledge_item=knowledge_item
        )
        
        if result and result[0].get("max_version"):
            return result[0]["max_version"]
        return 0
    
    async def _save_version_to_neo4j(self, version: KnowledgeVersion):
        """保存版本到Neo4j"""
        query = """
        CREATE (v:KnowledgeVersion {
            version_id: $version_id,
            knowledge_type: $knowledge_type,
            knowledge_item: $knowledge_item,
            version_number: $version_number,
            previous_version_id: $previous_version_id,
            created_at: $created_at,
            created_by: $created_by,
            change_summary: $change_summary,
            change_details: $change_details,
            source: $source,
            evidence_level: $evidence_level,
            status: $status,
            metadata: $metadata
        })
        """
        await self.neo4j.execute_query(
            query,
            version_id=version.version_id,
            knowledge_type=version.knowledge_type,
            knowledge_item=version.knowledge_item,
            version_number=version.version_number,
            previous_version_id=version.previous_version_id,
            created_at=version.created_at.isoformat(),
            created_by=version.created_by,
            change_summary=version.change_summary,
            change_details=version.change_details,
            source=version.source,
            evidence_level=version.evidence_level,
            status=version.status,
            metadata=version.metadata
        )
    
    async def get_version(self, version_id: str) -> Optional[KnowledgeVersion]:
        """获取版本"""
        query = """
        MATCH (v:KnowledgeVersion {version_id: $version_id})
        RETURN v
        """
        result = await self.neo4j.execute_query(query, version_id=version_id)
        
        if result:
            return KnowledgeVersion(**result[0]["v"])
        return None
    
    async def list_versions(self, knowledge_type: str, knowledge_item: int,
                           limit: int = 100) -> List[KnowledgeVersion]:
        """获取版本列表"""
        query = """
        MATCH (v:KnowledgeVersion)
        WHERE v.knowledge_type = $knowledge_type 
          AND v.knowledge_item = $knowledge_item
        RETURN v
        ORDER BY v.version_number DESC
        LIMIT $limit
        """
        result = await self.neo4j.execute_query(
            query,
            knowledge_type=knowledge_type,
            knowledge_item=knowledge_item,
            limit=limit
        )
        
        return [KnowledgeVersion(**record["v"]) for record in result]
    
    async def activate_version(self, version_id: str) -> bool:
        """激活版本"""
        query = """
        MATCH (v:KnowledgeVersion {version_id: $version_id})
        SET v.status = 'active'
        RETURN v
        """
        result = await self.neo4j.execute_query(query, version_id=version_id)
        return len(result) > 0
```

```python
# app/evolution/change_tracker.py
from typing import List, Dict, Any
from datetime import datetime
from app.models.evolution import KnowledgeChange, ChangeType
from app.utils.neo4j_client import Neo4jClient
from app.utils.logger import get_logger

logger = get_logger(__name__)

class ChangeTracker:
    """变更追踪器"""
    
    def __init__(self, neo4j_client: Neo4jClient):
        self.neo4j = neo4j_client
    
    def generate_change_id(self, version_id: str, entity_id: str, 
                          change_type: str) -> str:
        """生成变更ID"""
        timestamp = datetime.now().strftime("%Y%m%d%H%M%S")
        return f"CHANGE_{timestamp}_{entity_id}_{change_type}"
    
    async def record_change(self, version_id: str, change_type: ChangeType,
                           entity_type: str, entity_id: str,
                           old_value: Any, new_value: Any,
                           change_reason: str, operator: str,
                           field_name: Optional[str] = None,
                           change_source: str = "manual") -> KnowledgeChange:
        """记录变更"""
        try:
            change_id = self.generate_change_id(version_id, entity_id, change_type.value)
            
            change = KnowledgeChange(
                change_id=change_id,
                version_id=version_id,
                change_type=change_type,
                entity_type=entity_type,
                entity_id=entity_id,
                field_name=field_name,
                old_value=old_value,
                new_value=new_value,
                change_reason=change_reason,
                change_source=change_source,
                operator=operator
            )
            
            # 保存到Neo4j
            await self._save_change_to_neo4j(change)
            
            # 关联到版本
            await self._link_change_to_version(version_id, change_id)
            
            logger.info(f"记录变更: {change_id}")
            return change
            
        except Exception as e:
            logger.error(f"记录变更失败: {str(e)}")
            raise
    
    async def _save_change_to_neo4j(self, change: KnowledgeChange):
        """保存变更到Neo4j"""
        query = """
        CREATE (c:KnowledgeChange {
            change_id: $change_id,
            version_id: $version_id,
            change_type: $change_type,
            entity_type: $entity_type,
            entity_id: $entity_id,
            field_name: $field_name,
            old_value: $old_value,
            new_value: $new_value,
            change_reason: $change_reason,
            change_source: $change_source,
            timestamp: $timestamp,
            operator: $operator
        })
        """
        await self.neo4j.execute_query(
            query,
            change_id=change.change_id,
            version_id=change.version_id,
            change_type=change.change_type.value,
            entity_type=change.entity_type,
            entity_id=change.entity_id,
            field_name=change.field_name,
            old_value=str(change.old_value) if change.old_value else None,
            new_value=str(change.new_value) if change.new_value else None,
            change_reason=change.change_reason,
            change_source=change.change_source,
            timestamp=change.timestamp.isoformat(),
            operator=change.operator
        )
    
    async def _link_change_to_version(self, version_id: str, change_id: str):
        """关联变更到版本"""
        query = """
        MATCH (v:KnowledgeVersion {version_id: $version_id})
        MATCH (c:KnowledgeChange {change_id: $change_id})
        CREATE (v)-[:HAS_CHANGE]->(c)
        """
        await self.neo4j.execute_query(query, version_id=version_id, change_id=change_id)
    
    async def get_changes_by_version(self, version_id: str) -> List[KnowledgeChange]:
        """获取版本的变更记录"""
        query = """
        MATCH (v:KnowledgeVersion {version_id: $version_id})-[:HAS_CHANGE]->(c:KnowledgeChange)
        RETURN c
        ORDER BY c.timestamp ASC
        """
        result = await self.neo4j.execute_query(query, version_id=version_id)
        
        return [KnowledgeChange(**record["c"]) for record in result]
    
    async def get_entity_change_history(self, entity_id: str, 
                                       entity_type: str) -> List[KnowledgeChange]:
        """获取实体的变更历史"""
        query = """
        MATCH (c:KnowledgeChange)
        WHERE c.entity_id = $entity_id AND c.entity_type = $entity_type
        RETURN c
        ORDER BY c.timestamp DESC
        """
        result = await self.neo4j.execute_query(
            query,
            entity_id=entity_id,
            entity_type=entity_type
        )
        
        return [KnowledgeChange(**record["c"]) for record in result]
```

```python
        conflicts.extend(value_conflicts)
        
        # 检测关系冲突
        relationship_conflicts = await self._detect_relationship_conflicts(version_id1, version_id2)
        conflicts.extend(relationship_conflicts)
        
        # 检测逻辑冲突
        logic_conflicts = await self._detect_logic_conflicts(version_id1, version_id2)
        conflicts.extend(logic_conflicts)

        return conflicts

        MATCH (v2:KnowledgeVersion {version_id: $version_id2})-[:HAS_CHANGE]->(c2:KnowledgeChange)
        WHERE c1.entity_id = c2.entity_id
          AND c1.entity_type = c2.entity_type
          AND c1.field_name = c2.field_name
          AND c1.new_value <> c2.new_value
        RETURN c1.entity_id as entity_id,

                status=ConflictStatus.DETECTED
```

```python
router = APIRouter(prefix="/evolution", tags=["知识演化"])

evolution_service = EvolutionService()

@router.post("/version/create", response_model=KnowledgeVersion)
async def create_version(request: VersionCreateRequest):
    """创建新版本"""
    try:
        version = await evolution_service.create_version(
            knowledge_type=request.knowledge_type,
            knowledge_item=request.knowledge_item,
            source=request.source,
            change_summary=request.change_summary,
            created_by=request.created_by,
            evidence_level=request.evidence_level,
            previous_version_id=request.previous_version_id
        )
        return version
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/version/{version_id}", response_model=KnowledgeVersion)
async def get_version(version_id: str):
    """获取版本详情"""
    version = await evolution_service.get_version(version_id)
    if not version:
        raise HTTPException(status_code=404, detail="版本不存在")
    return version

@router.get("/version/list", response_model=List[KnowledgeVersion])
async def list_versions(knowledge_type: str, knowledge_item: int, limit: int = 100):
    """获取版本列表"""
    versions = await evolution_service.list_versions(knowledge_type, knowledge_item, limit)
    return versions

@router.post("/version/compare", response_model=VersionCompareResult)
async def compare_versions(request: VersionCompareRequest):
    """对比版本"""
    result = await evolution_service.compare_versions(
        request.version_id1,
        request.version_id2
    )
    return result

@router.get("/changes/{version_id}", response_model=List[KnowledgeChange])
async def get_changes(version_id: str):
    """获取变更记录"""
    changes = await evolution_service.get_changes_by_version(version_id)
    return changes

@router.post("/conflict/detect", response_model=List[KnowledgeConflict])
async def detect_conflicts(version_id1: str, version_id2: str):
    """检测冲突"""
    conflicts = await evolution_service.detect_conflicts(version_id1, version_id2)
    return conflicts

@router.post("/conflict/resolve")
async def resolve_conflict(conflict_id: str, resolution: str):
    """解决冲突"""
    success = await evolution_service.resolve_conflict(conflict_id, resolution)
    if not success:
        raise HTTPException(status_code=500, detail="解决冲突失败")
    return {"status": "success", "message": "冲突已解决"}
```
